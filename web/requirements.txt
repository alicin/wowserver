# Fully pinned, including transitive dependencies — this file is the lockfile.
#
# The repo pins the mysql image to a digest and refuses `latest` for the game image
# (deploy/docker-compose.yml, hosting.md 3.6). A portal whose dependency set changes
# under it on the next rebuild would be the one unpinned thing in the stack.
#
# Regenerate against the base image the Dockerfile actually uses, not the host Python:
#
#   printf 'fastapi\nuvicorn\njinja2\nitsdangerous\nPyMySQL\npython-multipart\nhttptools\nuvloop\n' > base.txt
#   docker run --rm -v "$PWD:/w" -w /w python:3.13-slim sh -c \
#     "pip install -q --dry-run --report /w/r.json -r base.txt >/dev/null && python -c \"
#   import json;[print(i['metadata']['name']+'=='+i['metadata']['version'])
#     for i in sorted(json.load(open('/w/r.json'))['install'],
#                     key=lambda d: d['metadata']['name'].lower())]\""
#
# Resolved 2026-08-08 on python:3.13-slim.
